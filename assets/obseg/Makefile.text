# text

# Every text bank is built from source. The English banks (E) come from
# $(OBSEG_DIR)/text/L*E.c; the PAL banks (P) are made from the same source by
# tools/brittext.py, which applies British spelling and PAL's reworded lines.
# So a new string only needs adding to the E source.

$(BUILD_DIR)/$(OBSEG_DIR)/text/%P.c: $(OBSEG_DIR)/text/%E.c tools/brittext.py
	@mkdir -p $(dir $@)
	python3 tools/brittext.py $< $@

$(BUILD_DIR)/$(OBSEG_DIR)/text/%.rz: $(BUILD_DIR)/$(OBSEG_DIR)/text/%.bin
	$(RZ_COMP) $< $@

$(BUILD_DIR)/$(OBSEG_DIR)/text/%.bin: $(BUILD_DIR)/$(OBSEG_DIR)/text/%.elf
	$(OBJCOPY) $< $@ -O binary

$(BUILD_DIR)/$(OBSEG_DIR)/text/%.elf: $(BUILD_DIR)/$(OBSEG_DIR)/text/%.o
	$(LD) -T assets/obseg/text/LnameX.ld -o $@ $<

$(BUILD_DIR)/$(OBSEG_DIR)/text/%.o: $(OBSEG_DIR)/text/%.c
	$(CC) -c $(CFLAGS) -o $@ $<

$(BUILD_DIR)/$(OBSEG_DIR)/text/%.o: $(BUILD_DIR)/$(OBSEG_DIR)/text/%.c
	$(CC) -c $(CFLAGS) -o $@ $<

# List of language codes
TEXT_LANGS := E P

# Compose all possible text file names
TEXT_ALL_RZ_FILES := $(foreach lang,$(TEXT_LANGS),$(foreach name,$(TEXTNAMELIST),$(BUILD_DIR)/$(OBSEG_DIR)/text/L$(name)$(lang).rz))

# Both builds carry the E and P banks; the game loads one set (language.c).
TEXT_RZ_FILES := $(TEXT_ALL_RZ_FILES)

# Default target
all: $(TEXT_RZ_FILES)

.PHONY: all
